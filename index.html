'usmash' : input.down ? 'dsmash' : 'fsmash', elapsed: 0 }; } else { this.startAttack(f, this.chooseAerial(f, input, dirH), onGround); } } else if (input.attackPressed) { const id = onGround ? this.chooseGround(f, input, dirH) : this.chooseAerial(f, input, dirH); this.startAttack(f, id, onGround); } } // --- スマッシュため --- if (f.charge) { f.charge.elapsed += dt; if (!input.smashHeld || f.charge.elapsed >= SMASH_MAX_CHARGE) { const power = 1 + (0.5 * Math.min(f.charge.elapsed, SMASH_MAX_CHARGE)) / SMASH_MAX_CHARGE; const id = f.charge.id; f.charge = null; this.startAttack(f, id, onGround, power); } } // --- 攻撃進行 --- if (f.attack) this.updateAttack(f, opp, dt, onGround); // --- 見た目 --- s.setFlipX(f.facing === -1); s.setBounce(f.hitstun > 0 ? 0.4 : 0, 0); // ふっとび中だけ壁・床でバウンド if (f.shieldStun > 0) s.setTint(Math.floor(f.shieldStun / 100) % 2 ? 0xffff66 : 0xff6666); else if (f.hitstun > 0) s.setTint(0xff9999); else if (f.charge) s.setTint(Math.floor(f.charge.elapsed / 60) % 2 ? 0xffffaa : 0xffffff); else if (f.helpless) s.setTint(0x777777); else s.clearTint(); s.setAlpha(f.invuln > 0 && Math.floor(f.invuln / 80) % 2 === 0 ? 0.35 : 1); f.shieldBubble.setVisible(f.shielding); if (f.shielding) { const base = Math.max(s.displayWidth, s.displayHeight) / 2 + 8; f.shieldBubble.setPosition(s.x, s.y); f.shieldBubble.setRadius(base * (0.45 + (0.55 * f.shieldHP) / 100)); } f.percentText.setPosition(s.x, body.top - 16); f.percentText.setText(`${Math.floor(f.damage)}%`); f.percentText.setColor( f.damage < 50 ? '#ffffff' : f.damage < 100 ? '#ffd166' : f.damage < 150 ? '#ff9f43' : '#ff4757' ); f.wasOnGround = onGround; } private chooseGround(f: Fighter, input: FighterInput, dirH: number): MoveId { if (input.up) return 'utilt'; if (input.down) return 'dtilt'; if (dirH !== 0) { f.facing = dirH as Dir; return 'ftilt'; } return 'jab'; } private chooseAerial(f: Fighter, input: FighterInput, dirH: number): MoveId { if (input.up) return 'uair'; if (input.down) return 'dair'; if (dirH === 0) return 'nair'; return dirH === f.facing ? 'fair' : 'bair'; } private doSpecial(f: Fighter, input: FighterInput, dirH: number, onGround: boolean) { if (input.up) { this.startAttack(f, 'upB', onGround); } else if (input.down) { this.startAttack(f, 'downB', onGround); } } // ------------------------------------------------------------------- // 攻撃処理 // ------------------------------------------------------------------- private startAttack(f: Fighter, id: MoveId, onGround: boolean, power: number = 1) { const move = MOVES[id]; // 踏み込み break; } } // 判定持続中 if (a.activated && a.elapsed < activeEnd) { if (a.id === 'sideB') body.setVelocity(a.dir * 520, 0); if (a.id === 'downB') body.setVelocityY(Math.max(body.velocity.y, 950)); } } } // 終了 if (a.elapsed >= activeEnd + m.endlag) { f.attack = null; if (m.helplessAfter && !onGround) f.helpless = true; } } private getHitboxRect(f: Fighter, m: MoveData, dir: Dir): Phaser.Geom.Rectangle { const s = f.sprite; } /** ヒット処理。シールドで防がれた場合もtrue（＝そのワザは当たり済み）を返す */ private applyHit(att: Fighter, tgt: Fighter, m: MoveData, dir: Dir, power: number): boolean { if (tgt.invuln > 0) return false; const P = this.params; const tBody = tgt.sprite.body as Body; const dmg = Math.max(1, Math.round(P.attackDamage * m.dmg * power)); // ガード if (tgt.shielding) { tgt.shieldHP -= dmg * 2.2; tBody.setVelocityX(dir * 160); soundManager.playShield(); this.showFloatingText(tgt.sprite.x, tgt.sprite.y - 30, 'GUARD', '#3498db'); if (tgt.shieldHP <= 0) this.breakShield(tgt); return true; } tgt.damage = Math.min(999, tgt.damage + dmg); // スマブラ式：基礎ふっとばし ＋ 成長率 × 蓄積% ÷ 重さ const kb = P.baseKnockback * m.bkb + (P.knockbackGrowth * m.kbg * tgt.damage * 100) / P.weight; const hDir = m.away ? ((Math.sign(tgt.sprite.x - att.sprite.x) || dir) as Dir) : dir; const rad = m.angle * DEG; const kbX = Math.cos(rad) * kb * hDir; let kbY = -Math.sin(rad) * kb; const tOnGround = tBody.blocked.down || tBody.touching.down; if (kbY > 0 && tOnGround) kbY = -kbY * 0.6; // 地上の相手へのメテオは打ち上げに tBody.setVelocity(kbX, kbY); tgt.hitstun = Phaser.Math.Clamp(kb * 0.3, 120, 1000); tgt.attack = null; tgt.charge = null; tgt.shielding = false; tgt.helpless = false; tgt.landingLag = 0; tgt.jumpsLeft = Math.max(tgt.jumpsLeft, 1); // 被弾で空中ジャンプが1回戻る soundManager.playHit(); this.blastEffectEmitter.emitParticleAt(tgt.sprite.x, tgt.sprite.y, kb > 900 ? 16 : 8); this.showFloatingText(tgt.sprite.x, tgt.sprite.y - 20, `+${dmg}%`); if (kb > 900) { this.cameras.main.shake(180, 0.012); this.showFloatingText(tgt.sprite.x, tgt.sprite.y - 50, 'SMASH!', '#ffa502', 24); } else { this.cameras.main.shake(80, 0.005); } return true; } private breakShield(f: Fighter) { f.shielding = false; f.shieldHP = 30; f.shieldStun = 1800; (f.sprite.body as Body).setVelocityY(-420); soundManager.playHit(); this.cameras.main.shake(150, 0.01); this.showFloatingText(f.sprite.x, f.sprite.y - 40, 'SHIELD BREAK!', '#f1c40f', 22); } private spawnProjectile(f: Fighter, dir: Dir) { const s = f.sprite; const img = this.physics.add.image(s.x + dir * (s.displayWidth / 2 + 10), s.y - 4, 'energy'); const b = img.body as Body; b.setAllowGravity(false); b.setVelocityX(dir * 520); img.setTint(f.color); f.projectile = { sprite: img, dir, life: 1100 }; } private updateProjectiles(dt: number) { for (const f of this.fighters) { const pr = f.projectile; } } if (done) { pr.sprite.destroy(); f.projectile = null; } } } // ------------------------------------------------------------------- // CPU AI // ------------------------------------------------------------------- private cpuReactionMs(): number { return Math.max(60, 480 - this.cpuLevel * 45) + Math.random() * 80; } private computeCpuInput(dt: number): FighterInput { const input = emptyInput(); const f = this.cpu; if (!f || this.cpuMode === 'sandbag') return input; const s = f.sprite; const body = s.body as Body; const onGround = body.blocked.down || body.touching.down; const ai = this.ai; ai.jumpCooldown -= dt; // スマッシュをため中 if (f.charge) { input.smashHeld = ai.smashHoldMs > 0; ai.smashHoldMs -= dt; return input; } // 復帰（場外・ステージより下にいるとき最優先） const st = this.getStageBounds(); if (!onGround && f.hitstun <= 0 && (s.x < st.left || s.x > st.right || s.y > st.top)) { const toCenter: Dir = s.x < this.params.gameWidth / 2 ? 1 : -1; } return input; } private decideCpuAction(f: Fighter, opp: Fighter, input: FighterInput, onGround: boolean) { const ai = this.ai; const lv = this.cpuLevel; ai.hold = { left: false, right: false, up: false, down: false }; input.specialPressed = true; return; } // 接近 hold(toward); if (!onGround && dy > 60) ai.hold.down = true; } // ------------------------------------------------------------------- // 撃墜 // ------------------------------------------------------------------- private isOutOfBlastZone(s: Phaser.Physics.Arcade.Sprite): boolean { const m = 180; } private showFloatingText(x: number, y: number, text: string, color: string = '#ff4757', size: number = 18) { const pop = this.add .text(x, y, text, { fontSize: `${size}px`, fontStyle: 'bold', color, stroke: '#ffffff', strokeThickness: 3, }) .setOrigin(0.5) .setDepth(15);
